rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Helper functions
    function isAuthenticated() {
      return request.auth != null;
    }

    // Public species catalog (Read-only for app users)
    match /species_catalog/{speciesId} {
      allow read: if isAuthenticated();
      allow write: if false; // Admin managed
    }

    // User-owned Garden Environments
    match /garden_environments/{envId} {
      allow read, write: if isAuthenticated() && resource.data.user_id == request.auth.uid;
      allow create: if isAuthenticated() && request.resource.data.user_id == request.auth.uid;
    }

    // User-owned Plants & Nested Subcollections
    match /plants/{plantId} {
      allow read, write: if isAuthenticated() && resource.data.user_id == request.auth.uid;
      allow create: if isAuthenticated() && request.resource.data.user_id == request.auth.uid;

      // Care logs & Health assessments inherit ownership validation
      match /{subcollection}/{docId} {
        allow read, write: if isAuthenticated() && get(/databases/$(database)/documents/plants/$(plantId)).data.user_id == request.auth.uid;
      }
    }
  }
}
